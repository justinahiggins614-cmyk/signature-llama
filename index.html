● DEVELOPERS

    
Drop the engine into any page. Same spirit as the backend's dial/chat calls: load, then generate.

    
    
      
  • 1 · Include the engine
        
    <script src="https://justinahiggins614-cmyk.github.io/signature-backend/sigllama/sigllama.js"></script>
    
        
    • 
      
  • 2 · Load the model
        
    await SigLlama.load('https://justinahiggins614-cmyk.github.io/signature-backend/sigllama/');
SigLlama.loaded(); // true when ready
SigLlama.info();   // { params, vocab, layers, ... }
    
        
    • 
      
  • 3 · Generate text
        
    const reply = await SigLlama.generate('The inventor opened the notebook and', {
  maxTokens: 80,
  temperature: 0.8,
  topK: 40
});
console.log(reply);
    
        
    • 
      
  • 4 · Chat loop (streaming tokens)
        
    const history = [];
async function chat(userText) {
  history.push('Human: ' + userText + '\nLlama:');
  let out = '';
  await SigLlama.generate(history.join('\n'), {
    maxTokens: 90, temperature: 0.85,
    onToken: t => { out += t; process.stdout.write(t); }
  });
  history.push(out);
  return out;
}
